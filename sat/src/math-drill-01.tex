\section*{Instructions}
Complete all 22 questions in approximately 35 minutes. A calculator is permitted. Select the best answer for each question. Give exact answers unless a question requests rounding. Difficulty ratings range from 1 (least difficult) to 5 (most difficult).

\section*{Problems}
\begin{enumerate}[leftmargin=*,itemsep=1.2em,topsep=0.8em]
  \item \textit{Difficulty: 4/5} A theater sold 420 tickets. Student tickets cost \$8, adult tickets cost \$14, and premium tickets cost \$25. Twice as many student tickets as premium tickets were sold, and total revenue was \$5,820. How many adult tickets were sold?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item 180 \item 210 \item 240 \item 300
  \end{enumerate}

  \item \textit{Difficulty: 3/5} The function $f(x)=x^2+bx+c$, where $b$ and $c$ are constants, satisfies $f(2)=3$ and $f(-1)=12$. What is $f(5)$?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item 7 \item 10 \item 12 \item 17
  \end{enumerate}

  \item \textit{Difficulty: 3/5} For what value of $k$ does the system have no solution?
  \[
    3x+ky=7,\qquad 6x+8y=15
  \]
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item 2 \item 4 \item $\dfrac{15}{2}$ \item 8
  \end{enumerate}

  \item \textit{Difficulty: 5/5} A company sells 600 bottles per month at \$24 each. For every \$2 increase in price, monthly sales are predicted to decrease by 30 bottles. Which price and monthly revenue maximize revenue under this model?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item \$28 and \$15,120 \item \$30 and \$15,300
    \item \$32 and \$15,360 \item \$34 and \$15,300
  \end{enumerate}

  \item \textit{Difficulty: 3/5} A culture initially contains 4,800 bacteria. For 3 hours, it increases by 25\% per hour. During each of the next 2 hours, it decreases by 20\% per hour. Approximately how many bacteria remain?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item 3,840 \item 6,000 \item 7,500 \item 9,375
  \end{enumerate}

  \item \textit{Difficulty: 5/5} Two observation points lie on the same level path leading to a vertical tower and are 40 meters apart. The angles of elevation to the top from the farther and nearer points are $30^\circ$ and $60^\circ$, respectively. What is the tower's height?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item $20$ m \item $20\sqrt3$ m \item $40$ m \item $40\sqrt3$ m
  \end{enumerate}

  \item \textit{Difficulty: 4/5} A right triangle has legs of lengths $x$ and $2x+2$ and hypotenuse $2x+3$, where $x>0$. What is $\sin\theta$ if $\theta$ is opposite the shorter leg?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item $\dfrac5{12}$ \item $\dfrac5{13}$ \item $\dfrac{12}{13}$ \item $\dfrac{13}{12}$
  \end{enumerate}

  \item \textit{Difficulty: 3/5} A circle has equation
  \[
    x^2+y^2-8x+6y-11=0.
  \]
  A square is circumscribed about the circle with every side tangent to it. What is the square's area?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item 36 \item 72 \item 144 \item 288
  \end{enumerate}

  \item \textit{Difficulty: 3/5} The graph of $y=ax^2+bx+c$ has vertex $(3,-8)$ and passes through $(1,4)$. What is $a+b+c$?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item 3 \item 4 \item 12 \item 19
  \end{enumerate}

  \item \textit{Difficulty: 4/5} A cylindrical tank is initially $\tfrac35$ full. After 144 gallons are removed, it is $\tfrac38$ full. Water is then added at 20 gallons per minute. How many minutes will it take to become $\tfrac78$ full?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item 12 \item 16 \item 18 \item 24
  \end{enumerate}

  \item \textit{Difficulty: 4/5} For $p(x)=2x^3+kx^2-5x+6$, where $k$ is constant, $p(2)=20$. What is the remainder when $p(x)$ is divided by $x+3$?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item $-69$ \item $-15$ \item 15 \item 69
  \end{enumerate}

  \item \textit{Difficulty: 3/5} A survey of 800 voters found 52\% support with a margin of error of 3 percentage points. A later survey quadruples the sample size. If margin of error varies inversely with the square root of sample size and observed support is 54\%, which interval should be reported?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item 51\% to 57\% \item 52.5\% to 55.5\%
    \item 53.25\% to 54.75\% \item 53.5\% to 54.5\%
  \end{enumerate}

  \item \textit{Difficulty: 3/5} A delivery service charges according to $C(n)=an+b$, where $n$ is the number of packages. Delivering 18 packages costs \$129, and delivering 30 costs \$189. What is the cost of delivering 45 packages?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item \$225 \item \$249 \item \$264 \item \$279
  \end{enumerate}

  \item \textit{Difficulty: 3/5} A container holds 30 liters of an 18\% salt solution. How many liters of pure water must be added to produce a 12\% salt solution?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item 10 \item 12 \item 15 \item 18
  \end{enumerate}

  \item \textit{Difficulty: 4/5} An exponential function $q(t)=ab^t$ satisfies $q(2)=18$ and $q(5)=486$. What is the value of $b$?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item 2 \item 3 \item 6 \item 9
  \end{enumerate}

  \item \textit{Difficulty: 3/5} An arc of a circle with radius 6 has length $5\pi$. What is the central angle subtending the arc, in radians?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item $\dfrac{5\pi}{12}$ \item $\dfrac{5\pi}{6}$
    \item $\dfrac{6\pi}{5}$ \item $\dfrac{5\pi}{3}$
  \end{enumerate}

  \item \textit{Difficulty: 4/5} For which value or values of $k$ does
  \[
    x^2+(k-3)x+k=0
  \]
  have exactly one real solution?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item 1 only \item 9 only \item 1 and 9 \item 3 and 10
  \end{enumerate}

  \item \textit{Difficulty: 4/5} A train travels 390 miles in 5.6 hours. It travels the first part at 60 miles per hour and the remaining part at 75 miles per hour. How many miles does it travel at 60 miles per hour?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item 90 \item 120 \item 150 \item 270
  \end{enumerate}

  \item \textit{Difficulty: 4/5} The function $f(x)=a(x-1)^2+2$ satisfies $f(3)=10$. If $h(x)=f(x+2)-3$, what is $h(4)$?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item 19 \item 31 \item 49 \item 51
  \end{enumerate}

  \item \textit{Difficulty: 2/5} Eight measurements have a mean of 14. After one measurement is removed, the remaining seven have a mean of 12. What measurement was removed?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item 14 \item 24 \item 26 \item 28
  \end{enumerate}

  \item \textit{Difficulty: 4/5} Two cones are similar. The smaller cone has height 6 and volume $24\pi$. The larger cone has height 15. What is the larger cone's volume?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item $60\pi$ \item $150\pi$ \item $240\pi$ \item $375\pi$
  \end{enumerate}

  \item \textit{Difficulty: 4/5} The equation
  \[
    \frac{x+a}{x-2}=3
  \]
  has solution $x=5$, where $a$ is constant. What is the solution of
  \[
    \frac{x+a}{x+1}=2?
  \]
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item 1 \item 2 \item 3 \item 4
  \end{enumerate}
\end{enumerate}
