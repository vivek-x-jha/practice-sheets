\section*{Instructions}
Complete 12 questions in 30 minutes. A calculator, including a graphing calculator, is permitted. Select exactly one answer (A--D) per question. All choices are exact; do not round intermediate calculations. All variables are real. Use the blank space for work.

\noindent\textit{Original, unofficial practice targeting harder-module reasoning. This is a challenge set, not a complete adaptive module or a score predictor.}

\section*{Questions}
\begin{enumerate}[leftmargin=*,itemsep=1em]
  \item For which value of $a$ does the following system have no solution?
  \[
    (a-2)x+3y=7,\qquad 6x+(a+1)y=14
  \]
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item $-4$ \item $-2$ \item $4$ \item $5$
  \end{enumerate}

  \vfill
  \item The quadratic function $f$ is defined by $f(x)=ax^2+bx+c$, where $a$, $b$, and $c$ are constants. If $f(2)=24$, $f(4)=-8$, and $f(8)=24$, what is the value of $a+b+c$?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item $-12$ \item $4$ \item $52$ \item $88$
  \end{enumerate}

  \vfill
  \item The function $g$ is defined by $g(x)=ab^x+c$, where $a>0$, $b>1$, and $c$ is a constant. The table gives three values of $g$.
  \[
    \begin{array}{c|ccc}
      x & 0 & 2 & 4 \\\hline
      g(x) & 11 & 35 & 131
    \end{array}
  \]
  What is $g(1)$?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item $17$ \item $19$ \item $23$ \item $47$
  \end{enumerate}

  \vfill
  \newpage
  \item A factory uses only production lines A and B. In one day's production, $4\%$ of the components from line A and $12\%$ of the components from line B were defective. Overall, $7\%$ of that day's components were defective. If a component is selected at random from that day's nondefective components, what is the probability that it came from line B?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item $\dfrac{7}{100}$ \item $\dfrac{3}{25}$
    \item $\dfrac{3}{8}$ \item $\dfrac{11}{31}$
  \end{enumerate}

  \vfill
  \item For each constant $k$, consider the equation
  \[
    \frac{x^2-9}{x-3}=kx.
  \]
  What is the sum of all values of $k$ for which the equation has no solution?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item $1$ \item $2$ \item $3$ \item $4$
  \end{enumerate}

  \vfill
  \item For each constant $k$, consider the equation
  \[
    (k-3)x^2+12x+3=0.
  \]
  What is the sum of all values of $k$ for which the equation has exactly one real solution?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item $3$ \item $12$ \item $15$ \item $18$
  \end{enumerate}

  \vfill
  \newpage
  \item A theater predicts that it will sell 240 tickets to a performance if each ticket costs \$30. For every \$3 increase in the ticket price, the theater predicts that it will sell 12 fewer tickets. The price may be increased only in \$3 increments. The cost of presenting the performance is \$1,800 plus \$6 for each ticket sold. Under this model, what is the greatest predicted profit, in dollars, from the performance?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item $3{,}960$ \item $5{,}220$ \item $5{,}256$ \item $8{,}064$
  \end{enumerate}

  \vfill
  \item Two right circular cylinders are similar. The ratio of the total surface area of the smaller cylinder to that of the larger cylinder is $9:25$. The larger cylinder's volume is 784 cubic centimeters greater than the smaller cylinder's volume. What is the sum of their volumes, in cubic centimeters?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item $980$ \item $1{,}216$ \item $1{,}568$ \item $3{,}136$
  \end{enumerate}

  \vfill
  \item In the $xy$-plane, the line $y=2x+1$ is tangent to the circle
  \[
    x^2+y^2-10x+6y+c=0,
  \]
  where $c$ is a constant. What is the value of $c$?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item $-\dfrac{196}{5}$ \item $-\dfrac{26}{5}$
    \item $\dfrac{26}{5}$ \item $\dfrac{196}{5}$
  \end{enumerate}

  \vfill
  \newpage
  \item In right triangle $ABC$, angle $C$ is a right angle. Point $D$ lies on hypotenuse $AB$, and $CD$ is perpendicular to $AB$. If $AD:DB=4:9$ and the area of triangle $ABC$ is 156 square units, what is the perimeter of triangle $ABC$, in units?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item $26+10\sqrt{13}$
    \item $26+20\sqrt{13}$
    \item $52+10\sqrt{13}$
    \item $52+20\sqrt{13}$
  \end{enumerate}

  \vfill
  \item For which positive value of $k$ does the equation
  \[
    \left|x^2-8x+7\right|=k
  \]
  have exactly three distinct real solutions?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item $7$ \item $8$ \item $9$ \item $16$
  \end{enumerate}

  \vfill
  \item If $x$ satisfies
  \[
    2^x+2^{-x}=5,
  \]
  what is the value of $2^{2x}+2^{-2x}$?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item $23$ \item $25$ \item $27$ \item $625$
  \end{enumerate}
  \vfill
\end{enumerate}
