\section*{Instructions}
Complete all 22 questions in 35 minutes. A calculator is permitted. Select the best answer for each question. The questions are original but modeled on the skills, structure, and multi-step reasoning of challenging official SAT Math questions.

\section*{Questions}
\begin{enumerate}[leftmargin=*,itemsep=1.15em]
  \item A linear function $f$ satisfies $f(3)=17$ and $f(8)=42$. What is $f^{-1}(67)$?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item 11 \item 12 \item 13 \item 15
  \end{enumerate}

  \item A laboratory purchased 42 sensors. Basic sensors cost \$48 each, and advanced sensors cost \$75 each. The total cost was \$2,556. How many basic sensors were purchased?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item 18 \item 20 \item 22 \item 24
  \end{enumerate}

  \item For what value of $k$ does the system below have no solution?
  \[
    (k-2)x+6y=9,
    \qquad
    4x+12y=25
  \]
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item 2 \item 4 \item 6 \item 8
  \end{enumerate}

  \item Plan A charges \$65 per month plus \$0.12 per text message. Plan B charges \$35 per month plus \$0.18 per text message. What is the least whole number of monthly text messages for which Plan A costs less than Plan B?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item 499 \item 500 \item 501 \item 502
  \end{enumerate}

  \item The line $Ax+By=26$ passes through $(3,2)$ and is perpendicular to $y=\frac23x-7$. What is $A+B$?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item 5 \item 8 \item 10 \item 13
  \end{enumerate}

  \item The polynomial $x^2+bx+c$ has a zero at $2+\sqrt3$, where $b$ and $c$ are rational constants. What is $b+c$?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item $-5$ \item $-3$ \item 1 \item 5
  \end{enumerate}

  \item Which expression is equivalent to
  \[
    \frac{x^2-9}{x^2+x-6}\cdot\frac{x+3}{x-3}
  \]
  for every value of $x$ in the domain of the original expression?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item $\dfrac{x+3}{x-2}$, where $x\ne-3,2,3$
    \item $\dfrac{x-3}{x-2}$, where $x\ne-3,2$
    \item $\dfrac{x+3}{x-2}$, where $x\ne2$
    \item $\dfrac{x+3}{x+2}$, where $x\ne-3,3$
  \end{enumerate}

  \item An exponential function $p(t)=ab^t$ satisfies $p(2)=54$ and $p(5)=1,458$. What is $p(4)$?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item 162 \item 324 \item 486 \item 972
  \end{enumerate}

  \item A rectangular enclosure borders a straight river, so fencing is needed on only three sides. If 240 meters of fencing is available, what is the greatest possible area of the enclosure?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item $3,600\text{ m}^2$ \item $7,200\text{ m}^2$
    \item $10,800\text{ m}^2$ \item $14,400\text{ m}^2$
  \end{enumerate}

  \item For $p(x)=x^3+kx^2-4x+7$, the remainder when $p(x)$ is divided by $x-2$ is 11. What is the remainder when $p(x)$ is divided by $x+1$?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item $-11$ \item $-1$ \item 7 \item 11
  \end{enumerate}

  \item The equation $\frac{x+3}{x-1}=k$ has solution $x=5$. What is the solution of
  \[
    \frac{2x-k}{x+2}=3?
  \]
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item $-8$ \item $-4$ \item 4 \item 8
  \end{enumerate}

  \item What is the solution of $\sqrt{2x+15}=x$?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item $-3$ \item 3 \item 5 \item 15
  \end{enumerate}

  \item The function $f$ has a minimum at $(-1,4)$. If $g(x)=-2f(x-3)+5$, at which point does $g$ have its corresponding maximum?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item $(-4,-3)$ \item $(2,-3)$ \item $(2,3)$ \item $(4,13)$
  \end{enumerate}

  \item The graphs of $y=x+2$ and $y=x^2-4x+8$ intersect at two points. What is the product of the $y$-coordinates of those points?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item 9 \item 12 \item 20 \item 25
  \end{enumerate}

  \item For each value of $k$, consider the equation
  \[
    x^2+(k-1)x+(k+5)=0.
  \]
  What is the sum of all values of $k$ for which the equation has exactly one real solution?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item $-6$ \item $-1$ \item 6 \item 19
  \end{enumerate}

  \item A container holds 80 liters of a solution that is 35\% acid. Some of the solution is removed and replaced by the same amount of pure water. The resulting solution is 28\% acid. How many liters were removed?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item 12 \item 16 \item 20 \item 28
  \end{enumerate}

  \item A data set has mean 12 and standard deviation 3. Every value $x$ is transformed to $2x-5$. What are the mean and standard deviation of the transformed data set?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item Mean 7; standard deviation 1
    \item Mean 19; standard deviation 3
    \item Mean 19; standard deviation 6
    \item Mean 24; standard deviation 6
  \end{enumerate}

  \item Of 120 students, 70 participate in music, 50 participate in a sport, and 30 participate in both. If a student who participates in a sport is selected at random, what is the probability that the student also participates in music?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item $\dfrac14$ \item $\dfrac37$ \item $\dfrac12$ \item $\dfrac35$
  \end{enumerate}

  \item A circle has center $(2,-3)$ and is tangent to the line $y=5$. When the circle's equation is written as
  \[
    x^2+y^2+Dx+Ey+F=0,
  \]
  what is $F$?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item $-64$ \item $-51$ \item 13 \item 51
  \end{enumerate}

  \item Two right circular cones are similar. The smaller cone has height 6, and the larger cone has height 15 and volume $1,000\pi$. What is the volume of the smaller cone?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item $64\pi$ \item $160\pi$ \item $400\pi$ \item $640\pi$
  \end{enumerate}

  \item An observer's eyes are 5 feet above level ground. The observer stands 30 feet from a building, and the tangent of the angle of elevation from the observer's eyes to the roof is $\frac43$. How tall is the building?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item 35 ft \item 40 ft \item 45 ft \item 50 ft
  \end{enumerate}

  \item In right triangle $ABC$, $\angle C=90^\circ$, the hypotenuse has length 26, and $\sin A=\frac5{13}$. An altitude is drawn from $C$ to the hypotenuse. What is the altitude's length?
  \begin{enumerate}[label=\Alph*.,itemsep=0pt]
    \item $\dfrac{60}{13}$ \item $\dfrac{120}{13}$
    \item 10 \item 12
  \end{enumerate}
\end{enumerate}
