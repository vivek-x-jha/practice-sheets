\section*{Instructions}

\section*{Problem}
Find the exact value of
\[
  \prod_{k=0}^{5}
  \left(
    \cos\left(\frac{k\pi}{6}\right)
    +
    \sin\left(\frac{k\pi}{6}\right)
  \right).
\]
