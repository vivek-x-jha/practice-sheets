\section*{Instructions}
Solve each equation over the real numbers. Show your work on separate paper
and give all solutions in exact form. Check each proposed solution in the
original equation; if none work, write ``no real solution.'' Work without a
calculator. Aim to finish in 40 minutes.

\section*{Problems}
\begin{enumerate}[leftmargin=*, itemsep=1.1em, topsep=0.8em]
  \item $\sqrt{3x+4}=7$
  \item $2\sqrt{x+5}-3=9$
  \item $\sqrt[3]{4x-7}+2=-1$
  \item $\sqrt{5x+6}=x$
  \item $\sqrt{2x+9}=x-3$
  \item $\sqrt{4x+5}=x-1$
  \item $\sqrt{3x+7}=\sqrt{x+1}+2$
  \item $\sqrt{x+5}+\sqrt{x-3}=4$
  \item $\sqrt{2x+3}-\sqrt{x-1}=2$
  \item $\sqrt{5-x}=x-6$
  \item $\sqrt{x^2-5x+10}=2$
  \item $\sqrt{x+\sqrt{4x-4}}=3$
\end{enumerate}
