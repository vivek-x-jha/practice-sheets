\section*{Instructions}
Complete all 12 problems without a calculator. Show the conversion steps and
use exact values. Problems 1--6 require Cartesian equations in $x$ and $y$;
Problems 7--12 require polar equations in $r$ and $\theta$.
Simplify and include any necessary restrictions. Unless directed to solve for
$r$, an equivalent factored polar equation is acceptable.

Allow signed values of $r$, with $0\le\theta<2\pi$.
Preserve the entire curve, including the origin when it belongs to the
original equation. If you divide by $r$, check the origin separately.

\section*{Reference}
\[
  x=r\cos\theta,\qquad y=r\sin\theta,\qquad r^2=x^2+y^2.
\]
\[
  \sin(2\theta)=2\sin\theta\cos\theta,\qquad
  \cos(2\theta)=\cos^2\theta-\sin^2\theta.
\]
For $r\ne0$, $\cos\theta=x/r$ and $\sin\theta=y/r$.
Also, $\sqrt{x^2+y^2}=|r|$, not necessarily $r$.

\section*{Polar to Cartesian}
\begin{enumerate}[leftmargin=*,itemsep=0pt]
  \item Convert $r=6\cos\theta-8\sin\theta$.
  Also state the center and radius of the resulting circle.
  \par\vspace{\stretch{1}}

  \item $r^2=3\cos\theta-2\sin\theta$
  \par\vspace{\stretch{1}}

  \item $r^2=8\cos(2\theta)$
  \par\vspace{\stretch{1}}

  \item $\displaystyle r=\frac{4}{1+\sin\theta}$
  \par\vspace*{\stretch{1}}
\end{enumerate}

\section*{Cartesian to Polar}
\begin{enumerate}[leftmargin=*,itemsep=0pt,resume]
  \item Convert $(x-3)^2+(y+2)^2=13$. Give an equation with $r$ isolated.
  \par\vspace{\stretch{1}}

  \item Convert $x^2+y^2+4x-6y-12=0$.
  Solve the resulting equation for $r$, including both real branches.
  \par\vspace*{\stretch{1}}
\end{enumerate}
