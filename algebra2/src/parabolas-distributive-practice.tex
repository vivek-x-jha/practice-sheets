\section*{Instructions}
Complete in 30 minutes. Show all work. Give exact answers unless a decimal is explicitly requested.

\section*{Problems}
\begin{enumerate}[leftmargin=*, itemsep=1.0em, topsep=0.8em]
  \item Expand and simplify:
  \[
  3(2x-5)^2-4(x+1)(x-6).
  \]

  \item Expand and simplify:
  \[
  (x-3)(2x+5)(x+4).
  \]

  \item Factor completely:
  \[
  4x^3-16x^2-9x+36.
  \]

  \item Factor completely:
  \[
  2x^4-13x^2-7.
  \]

  \item Solve:
  \[
  5(2x-1)^2-3(x+4)=2x^2+17.
  \]

  \item Solve:
  \[
  (x-2)(x+5)=3(2x-1).
  \]

  \item For $f(x)=2x^2-12x+5$, find the vertex, axis of symmetry, and minimum value.

  \item Write $g(x)=-3x^2+18x-20$ in vertex form, then state the maximum value.

  \item For $h(x)=x^2-4x-k$, find the value(s) of $k$ so the parabola has exactly one $x$-intercept.

  \item A parabola has $x$-intercepts $-2$ and $6$ and passes through $(1,-18)$. Write its equation in standard form.

  \item A rectangle has perimeter $40$ units. Write its area as a quadratic function of its width, then find the maximum possible area.

  \item A ball's height is modeled by $H(t)=-16t^2+72t+4$. Find the maximum height and the time when it occurs.
\end{enumerate}
