\section*{Instructions}
Complete in 40 minutes. Show clear work. Give exact values unless a rounding direction is stated. Use degrees or radians as shown.

\section*{Problems}
\begin{enumerate}[leftmargin=*,itemsep=1em,topsep=0.8em]
  \item Give the unit-circle coordinate $(\cos\theta,\sin\theta)$ for each angle:
  \[
  \frac{5\pi}{6},\qquad \frac{7\pi}{4},\qquad 240^\circ,\qquad -\frac{\pi}{3}.
  \]

  \item Evaluate exactly:
  \[
  \sin\frac{11\pi}{6},\quad
  \cos\frac{3\pi}{4},\quad
  \tan\frac{2\pi}{3},\quad
  \sec\frac{5\pi}{3},\quad
  \csc\frac{7\pi}{6},\quad
  \cot\frac{3\pi}{4}.
  \]

  \item A point on the unit circle is $\left(-\frac{12}{13},\frac{5}{13}\right)$. State the quadrant and find all six trigonometric function values.

  \item In a right triangle, relative to angle $\theta$, the opposite leg is $9$ and the adjacent leg is $40$. Find all six trigonometric function values.

  \item A right triangle has an acute angle of $35^\circ$ and hypotenuse $18$. Find the opposite and adjacent side lengths to the nearest tenth.

  \item From a point $28$ feet from the base of a vertical building, the angle of elevation to the roof is $52^\circ$. Find the building's height to the nearest tenth of a foot.

  \item Suppose $\sin\theta=\frac35$ and $\cos\theta=-\frac45$. Evaluate:
  \[
  \sin(\theta+4\pi),\quad
  \cos(\theta-6\pi),\quad
  \tan(\theta+5\pi),
  \]
  \[
  \sec(\theta-2\pi),\quad
  \csc(\theta+8\pi),\quad
  \cot(\theta-3\pi).
  \]

  \item State the period of each function:
  \[
  \sin(3x),\quad
  \cos\left(\frac{x}{4}\right),\quad
  \tan(5x),\quad
  \cot\left(\frac{2x}{3}\right),\quad
  \sec(2x),\quad
  \csc\left(\frac{x}{3}\right).
  \]

  \item A right triangle has hypotenuse $25$ and one leg $7$. Find the missing leg. If $\theta$ is opposite the missing leg, find $\sin\theta$, $\cos\theta$, and $\tan\theta$.

  \item If $\cos\theta=-\frac{5}{13}$ and $\theta$ is in Quadrant III, find
  \[
  \sin\theta,\qquad \tan\theta,\qquad \sec\theta,\qquad \csc\theta,\qquad \cot\theta.
  \]

  \item Find each missing value using a Pythagorean identity and the stated quadrant.
  \begin{enumerate}[label=(\alph*)]
    \item $\tan\alpha=\frac{7}{24}$ and $\alpha$ is in Quadrant I; find $\sec\alpha$.
    \item $\csc\beta=-\frac53$ and $\beta$ is in Quadrant IV; find $\cot\beta$.
  \end{enumerate}

  \item Complete or evaluate each statement:
  \begin{enumerate}[label=(\alph*)]
    \item $\sin 28^\circ=\cos(\underline{\hspace{1cm}}^\circ)$
    \item $\tan 17^\circ=\cot(\underline{\hspace{1cm}}^\circ)$
    \item $\sec 41^\circ=\csc(\underline{\hspace{1cm}}^\circ)$
    \item $\displaystyle \cos\left(\frac\pi2-\frac\pi6\right)$
    \item If $\theta$ is acute and $\sin\theta=\cos23^\circ$, find $\theta$.
  \end{enumerate}
\end{enumerate}

\section*{Answer Key}
\begin{enumerate}[leftmargin=*,itemsep=0.8em]
  \item $\left(-\frac{\sqrt3}{2},\frac12\right)$, $\left(\frac{\sqrt2}{2},-\frac{\sqrt2}{2}\right)$, $\left(-\frac12,-\frac{\sqrt3}{2}\right)$, $\left(\frac12,-\frac{\sqrt3}{2}\right)$

  \item $-\frac12$, $-\frac{\sqrt2}{2}$, $-\sqrt3$, $2$, $-2$, $-1$

  \item Quadrant II;
  \[
  \sin\theta=\frac5{13},\quad \cos\theta=-\frac{12}{13},\quad
  \tan\theta=-\frac5{12},\quad \csc\theta=\frac{13}{5},\quad
  \sec\theta=-\frac{13}{12},\quad \cot\theta=-\frac{12}{5}.
  \]

  \item The hypotenuse is $41$;
  \[
  \sin\theta=\frac9{41},\quad \cos\theta=\frac{40}{41},\quad
  \tan\theta=\frac9{40},\quad \csc\theta=\frac{41}{9},\quad
  \sec\theta=\frac{41}{40},\quad \cot\theta=\frac{40}{9}.
  \]

  \item Opposite $\approx10.3$; adjacent $\approx14.7$.

  \item $\approx35.8$ feet.

  \item $\frac35$, $-\frac45$, $-\frac34$, $-\frac54$, $\frac53$, $-\frac43$

  \item $\frac{2\pi}{3}$, $8\pi$, $\frac\pi5$, $\frac{3\pi}{2}$, $\pi$, $6\pi$

  \item Missing leg $24$; $\sin\theta=\frac{24}{25}$, $\cos\theta=\frac7{25}$, $\tan\theta=\frac{24}{7}$.

  \item $\sin\theta=-\frac{12}{13}$, $\tan\theta=\frac{12}{5}$, $\sec\theta=-\frac{13}{5}$, $\csc\theta=-\frac{13}{12}$, $\cot\theta=\frac5{12}$.

  \item (a) $\sec\alpha=\frac{25}{24}$ \qquad (b) $\cot\beta=-\frac43$

  \item (a) $62$ \quad (b) $73$ \quad (c) $49$ \quad (d) $\frac12$ \quad (e) $67^\circ$
\end{enumerate}
